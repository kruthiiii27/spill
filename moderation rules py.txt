from datetime import datetime, timedelta

def evaluate_user_enforcement(user, violation_severity: str):
    """
    Progressive enforcement matrix:
    - WARNING: increment warning count, block post submission
    - SEVERE: immediate restriction or temporary suspension
    - Escalations:
        2 warnings -> Posting restricted (24 hrs)
        3 warnings -> Suspended (7 days)
        4+ violations / Critical -> Permanent Ban
    """
    now = datetime.utcnow()
    user.last_violation_at = now
    
    if violation_severity == "WARNING":
        user.warning_count += 1
        if user.warning_count >= 2 and not user.is_suspended:
            user.status = "RESTRICTED"
        return "WARNING_ISSUED"

    elif violation_severity == "SEVERE":
        user.violation_count += 1
        if user.violation_count == 1:
            user.status = "RESTRICTED"
            action = "RESTRICTED"
        elif user.violation_count == 2:
            user.status = "SUSPENDED"
            user.is_suspended = True
            action = "SUSPENDED"
        else:
            user.status = "BANNED"
            user.is_banned = True
            action = "BANNED"
        return action
        
    return "NONE"
